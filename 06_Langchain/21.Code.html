LangChain ReAct Agent - Medical Query Routing (One Page)

# 1. IMPORTS
from langchain_openai import AzureChatOpenAI
from langchain_core.tools import tool
from langchain_core.prompts import PromptTemplate
from langchain.agents import create_react_agent, AgentExecutor
from langchain.agents.output_parsers import ReActSingleInputOutputParser

# 2. TOOLS - Web Search & PubMed Search
@tool
def web_search(query: str) -> str:
    """Search web for non-medical info"""
    return f"Web results for: {query}"

@tool
def pubmed_search(query: str) -> str:
    """Search PubMed for medical info"""
    return f"PubMed results for: {query}"

tools = [web_search, pubmed_search]

# 3. AZURE OPENAI CONFIGURATION
llm = AzureChatOpenAI(
    azure_endpoint="https://YOUR-RESOURCE.openai.azure.com/",
    api_key="YOUR_API_KEY", api_version="2024-10-21",
    azure_deployment="gpt-4.1", temperature=0
)

# 4. PROMPT TEMPLATE WITH ROUTING LOGIC
prompt = PromptTemplate.from_template("""You are intelligent agent with 2 tools: {tools}
Tool names: {tool_names}

ROUTING:
- Medicine/health/disease/symptoms/drugs/treatment/clinical → pubmed_search
- All other questions → web_search
Format:
Question: {input}
Thought: (decide tool)
Action: [web_search|pubmed_search]
Action Input: (search query)
Observation: (result)
Final Answer: (response)

{agent_scratchpad}""")

# 5. CREATE & EXECUTE AGENT
output_parser = ReActSingleInputOutputParser()
agent = create_react_agent(llm=llm, tools=tools, prompt=prompt, output_parser=output_parser)
agent_executor = AgentExecutor(agent=agent, tools=tools, verbose=True)

# 6. INVOKE AGENT WITH QUERY
response = agent_executor.invoke({"input": "What are symptoms of diabetes?"})
